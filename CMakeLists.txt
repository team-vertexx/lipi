cmake_minimum_required(VERSION 3.16)
project(lipi LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
add_executable(lipi src/main.cpp src/parse.cpp src/ocsf.cpp)
target_compile_options(lipi PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-pedantic -O3)
target_link_libraries(lipi PRIVATE Threads::Threads)
